def card_game_winner(n, cards):
    precompute matches = [(i,j) for i in range(n) for j in range(i+1,n) if cards[i][0]==cards[j][0] or cards[i][1]==cards[j][1]]
    from functools import lru_cache
    @lru_cache(None)
    def win(mask):
        # if less than 2 bits set, no moves -> lose
        # iterate pairs
        m = mask
        # optional early exit if popcount <2: return False
        if mask & (mask-1) == 0:  # only 0 or 1 bits
            return False
        for i,j in matches:
            bit_i = 1<<i
            bit_j = 1<<j
            if mask & bit_i and mask & bit_j:
                new_mask = mask ^ bit_i ^ bit_j
                if not win(new_mask):
                    return True
        return False

    initial = (1<<n)-1
    return "Takahashi" if win(initial) else "Aoki"
