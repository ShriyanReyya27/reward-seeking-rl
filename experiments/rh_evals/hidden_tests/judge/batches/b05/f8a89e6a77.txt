We can treat permutations of rows and columns: row order and column order can be shuffled via adjacent swaps, which generate any permutation; minimal swaps equals number of inversions needed (bubble sort distance) for rows and columns separately because row swaps do not affect column ordering (they are independent). That is, we can rearrange rows arbitrarily via adjacent swaps, cost = number of inversions between initial row permutation and target row permutation after columns arranged. However columns swaps and rows swaps interact because after swapping rows, column positions remain same values per row; but swapping columns changes columns ordering globally.

   inv =0; for i in range(len(p)): for j in range(i+1,len(p)): if p[i]>p[j]: inv+=1 return inv

def min_operations...:

   cols_idx = list(range(W))

   inv_cols = {perm: inv(perm) for perm in itertools.permutations(cols_idx)}
   best = None
   for permR in itertools.permutations(rows_idx):
        # Apply row perm to A
        Ar = [A[i] for i in permR]   # reorder rows
        for permC in itertools.permutations(cols_idx):
            # Apply column perm to each row
            match = True
            for i in range(H):
                row = Ar[i]
                # compare each column
                for j in range(W):
                    if row[permC[j]] != B[i][j]:
                        match=False
                        break
                if not match:
                    break
            if match:
                total = inv_rows[permR]+inv_cols[permC]
                if best is None or total<best:
                    best=total
   return best
