```
def hungarian(cost):
    n = len(cost)
    m = len(cost[0])
    u = [0]*(n+1)
    v = [0]*(m+1)
    p = [0]*(m+1)
    way = [0]*(m+1)
    for i in range(1,n+1):
        p[0] = i
        minv = [INF]*(m+1)
        used = [False]*(m+1)
        j0 = 0
        while True:
            used[j0] = True
            i0 = p[j0]
            delta = INF
            j1 = 0
            for j in range(1,m+1):
                if not used[j]:
                    cur = cost[i0-1][j-1] - u[i0] - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            for j in range(m+1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta
            j0 = j1
            if p[j0]==0:
                break
        while True:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0==0:
                break
    # matching result: p[j] = row assigned to column j
    assignment = [0]* (n+1)
    for j in range(1,m+1):
        assignment[p[j]] = j
    result = sum(cost[i-1][assignment[i]-1] for i in range(1,n+1))
    return result
```
