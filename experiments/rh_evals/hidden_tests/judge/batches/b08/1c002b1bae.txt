def cover(bits_mask, avail_indices, limit):
    if bits_mask == 0:
        return set()
    # pick a bit (lowest set)
    b = (bits_mask & -bits_mask).bit_length()-1  # bit position
    candidates = [i for i in bit_to_indices[b] if i in avail_indices]
    # sort candidates by number of needed bits they cover descending
    candidates.sort(key=lambda i: bin(nums[i] & bits_mask).count('1'), reverse=True)
    for i in candidates[:10]:  # limit candidates
        new_selected = {i}
        new_bits_mask = bits_mask & ~ (nums[i] & bits_mask)
        # recursively cover remaining bits, with avail minus i, limit-1
        if limit-1 < 0: continue
        sub = cover(new_bits_mask, avail_indices - {i}, limit-1)
        if sub is not None:
            return new_selected | sub
    return None
