def sum_absolute_differences(n,a,b,queries):
    results=[]
    for x,y in queries:
        Bsub = b[:y]
        Bsorted = sorted(Bsub)
        prefix = [0]
        for val in Bsorted: prefix.append(prefix[-1]+val)
        totalY = prefix[-1]
        s=0
        for i in range(x):
            v = a[i]
            idx = bisect_right(Bsorted, v)
            cnt_le = idx
            sum_le = prefix[idx]
            cnt_gt = y - cnt_le
            sum_gt = totalY - sum_le
            s += cnt_le * v - sum_le + sum_gt - cnt_gt * v
        results.append(s)
    return results
