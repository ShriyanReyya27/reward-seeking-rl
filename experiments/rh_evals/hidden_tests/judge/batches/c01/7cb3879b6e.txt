def divide_into_good_sequences(L,R):
    intervals = []
    while L < R:
        remaining = R - L
        if L == 0:
            lowbit = 1 << (remaining.bit_length())  # >= remaining
        else:
            lowbit = L & -L
        block = lowbit
        # reduce block if too big
        while block > remaining:
            block >>=1
        intervals.append((L, L+block))
        L += block
    return intervals
